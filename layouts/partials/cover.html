{{- /* Site override of the theme partial. Adds an optional `coverAnimated` front matter param, used only when the page is being rendered on its own. Listings and the og:image still get the still `cover`, so nothing heavy loads on the home page. An `.mp4` or `.webm` renders as a muted looping video with the still image as its poster; anything else renders as an image. `page` is the page the template is rendering; `.` is the page the cover belongs to. They differ inside a listing loop, which is what distinguishes the two cases. */ -}} {{- $cover := false -}} {{- $autoCover := default $.Site.Params.autoCover false }} {{- if index .Params "cover" -}} {{- if .Resources.GetMatch .Params.Cover }} {{- $cover = (.Resources.GetMatch .Params.Cover).RelPermalink -}} {{- else -}} {{- $cover = absURL .Params.Cover -}} {{- end -}} {{- else if $.Site.Params.AutoCover -}} {{- if (not .Params.Cover) -}} {{- if .Resources.GetMatch "cover.*" -}} {{- $cover = (.Resources.GetMatch "cover.*").RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- $animated := false -}} {{- if and (index .Params "coverAnimated") (eq page.RelPermalink .RelPermalink) -}} {{- if .Resources.GetMatch .Params.coverAnimated }} {{- $animated = (.Resources.GetMatch .Params.coverAnimated).RelPermalink -}} {{- else -}} {{- $animated = absURL .Params.coverAnimated -}} {{- end -}} {{- end -}} {{- $isVideo := and $animated (in (slice "mp4" "webm") (path.Ext $animated | strings.TrimPrefix ".")) -}} {{- if $isVideo -}} {{- else -}} {{- with (or $animated $cover) -}} {{- end -}} {{- end }}
